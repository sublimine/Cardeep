## SQL BORRADOR (por tabla; [NO-VERIFICADO] — afinar con `\d` real de cada *_run)
```sql
-- 0070_vam_verified_needs_proof.sql  [BORRADOR — NO aplicar sin dry-run :5434]
ALTER TABLE canonical_dedup_run ADD COLUMN IF NOT EXISTS vam_verdict_id BIGINT
  REFERENCES verification_verdict(id);
CREATE OR REPLACE FUNCTION cardeep_vam_verified_needs_proof() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE v_verdict text; v_quorum int;
BEGIN
  IF NEW.vam_verified AND (TG_OP='INSERT' OR OLD.vam_verified IS DISTINCT FROM TRUE) THEN
    IF NEW.vam_verdict_id IS NULL THEN
      RAISE EXCEPTION 'run % cannot be vam_verified without vam_verdict_id', NEW.cluster_run_id
        USING ERRCODE='restrict_violation';
    END IF;
    SELECT verdict, quorum_n INTO v_verdict, v_quorum
      FROM verification_verdict WHERE id = NEW.vam_verdict_id;
    IF v_verdict IS DISTINCT FROM 'TRUSTWORTHY' OR COALESCE(v_quorum,0) < 2 THEN
      RAISE EXCEPTION 'run % vam_verified needs TRUSTWORTHY verdict quorum_n>=2 (got %/%)',
        NEW.cluster_run_id, COALESCE(v_verdict,'<missing>'), COALESCE(v_quorum,0)
        USING ERRCODE='restrict_violation';
    END IF;
  END IF;
  RETURN NEW;
END $$;
```
